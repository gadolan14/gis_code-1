  select(iso3, country, gii_2010, gii_2019) %>%

Join GII Data with World Map

library(rnaturalearth)

world <- ne_countries(
  scale = "medium",
  returnclass = "sf"
)

world_gii <- world %>%
  left_join(
    gii_selected,
    by = c("iso_a3" = "iso3")
  )

sum(!is.na(world_gii$gii_change))
## [1] 152

Map of Global GII Change

ggplot(world_gii) +
  geom_sf(aes(fill = gii_change),
          color = "white",
          linewidth = 0.1) +
  scale_fill_gradient2(
    low = "#2166ac",
    mid = "white",
    high = "#b2182b",
    midpoint = 0,
    na.value = "grey85",
    name = "GII Change"
  ) +
  labs(
    title = "Global Changes in Gender Inequality (2010–2019)",
    subtitle = "Change in GII = 2019 value - 2010 value",
    caption = "Source: UNDP Human Development Report 2025"
  ) +
  theme_minimal()

## Interpretation

The map shows changes in the Gender Inequality Index (GII) between 2010 and 2019. Negative values indicate a reduction in gender inequality, while positive values indicate an increase. Among the countries represented in the spatial dataset, 152 have valid GII change values. Countries with missing data are shown in grey.

The results suggest that gender inequality declined in many countries during this period, although the magnitude of change varies geographically.
